# The desktop session's scene corpus (desktop session).
#
# SceneCorpusTests.cpp renders every case with the compositor's Scene and
# Tools/Vm/AsteriaVm/Session.py with the runner's model; both check the PPM's
# SHA-256 against SceneDigests.txt, so their images are byte-identical.
# Lexical rules: Tests/Support/RenderCorpus.hpp.
#
#   case scene <name> <width> <height> <rgbx|bgrx>
#   font <vegur|fallback>              (default Vegur Bold/Regular UI)
#   window <x> <y> <width> <height> <focused|unfocused|attention> [flag...]
#   window maximized <focused|unfocused|attention> [flag...]   (the frame is the output)
#       flags: notresponding volume (a volume bundle) close-pressed maximize-pressed
#       windows stack in the order given, the first at the bottom
#   title "<name>" "<subtitle>"           (the last window's; a name of up to 16 bytes)
#   image pattern <seed> <width> <height> (the last window's image)
#   image solid <red> <green> <blue> <width> <height>
#   image terminal <terminal corpus case> (Tests/User/Terminal/TerminalCorpus.txt)
#   image none
#   switcher "<text>"
#   outline <x> <y> <width> <height>
#   end
